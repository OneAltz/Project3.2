int load_game(const char *filename, double *bank_balance)
{
    FILE *file = fopen(filename, "r");
    if (file == NULL) {
        printf("Could not load game");//perror?
        return 0;
    }

    double loaded_balance;
    if (fscanf(file, "%lf", &loaded_balance) != 1 || loaded_balance < 0) {
        fprintf(stderr, "Game file is invalid.\n");
        fclose(file);
        return 0;
    }

    fclose(file);
    *bank_balance = loaded_balance;
    return 1;
}
